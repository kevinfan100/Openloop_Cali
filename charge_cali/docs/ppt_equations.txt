% --- 1. 磁場 (Coulomb's law, μ₀ cancels) ---

B(x) = \frac{N_c \cdot I}{4\pi \cdot R_a \cdot (x+b)^2}


% --- 2. 磁場梯度 ---

\left|\frac{dB}{dx}\right| = \frac{2 \, N_c \cdot I}{4\pi \cdot R_a \cdot (x+b)^3}


% --- 3. 順磁球能量 → 力量 ---

U = -\frac{\chi_{\text{eff}} \cdot V_{\text{bead}}}{2\mu_0} \cdot B^2

F = -\frac{dU}{dx} = \frac{\chi_{\text{eff}} \cdot V_{\text{bead}}}{\mu_0} \cdot B \cdot \left|\frac{dB}{dx}\right|


% --- 4. 代入 B 和 dB/dx → 最終力量公式 (x, b in meters) ---

F(x) = \frac{2 \chi_{\text{eff}} V_{\text{bead}} N_c^2 I^2}{\mu_0 (4\pi)^2 R_a^2 (x+b)^5}


% --- 5. 雙線性形式 ---

F = \psi(x) \cdot I^2

\psi(x) = \frac{2 \chi_{\text{eff}} V_{\text{bead}} N_c^2}{\mu_0 (4\pi)^2 R_a^2 (x+b)^5}


% --- 6. R_a 從擬合反推 ---

a^2 = \frac{S_H \cdot N_c \cdot k_A}{4\pi \cdot R_a} \times 10^{12}

R_a = \frac{S_H \cdot N_c \cdot k_A \times 10^{12}}{4\pi \cdot a^2}


% --- 7. 球退磁修正 ---

\chi_{\text{eff}} = \frac{3\chi}{3 + \chi}

V_{\text{bead}} = \frac{4}{3}\pi r_{\text{bead}}^3
